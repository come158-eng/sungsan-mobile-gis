cmake_minimum_required(VERSION 3.20)
project(MetaThemeRegression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
enable_testing()
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
# Compile the unchanged production class with only the platform probe isolated.
configure_file("${REPO}/src/core/theme.cpp" theme.cpp COPYONLY)
configure_file("${REPO}/src/core/theme.h" theme.h COPYONLY)
configure_file(platformutilities.h platformutilities.h COPYONLY)
configure_file(qfield_core_export.h qfield_core_export.h COPYONLY)
foreach(brand IN ITEMS metaengi sungsan)
  set(THEME_JSON "${REPO}/branding/${brand}/theme.json")
  configure_file(theme.qrc.in "${brand}.qrc" @ONLY)
  add_executable(${brand}_theme main.cpp "${CMAKE_CURRENT_BINARY_DIR}/theme.cpp" "${CMAKE_CURRENT_BINARY_DIR}/${brand}.qrc")
  target_include_directories(${brand}_theme PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(${brand}_theme PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
  add_test(NAME ${brand}_appearance COMMAND ${brand}_theme ${brand})
  set_tests_properties(${brand}_appearance PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endforeach()
